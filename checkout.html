// --- DOM Elements --- const checkoutForm = document.getElementById('checkoutForm'); const checkoutItemsContainer = document.getElementById('checkout-items'); const subtotalDisplay = document.getElementById('subtotalDisplay'); const totalAmountDisplay = document.getElementById('totalAmountDisplay'); // Removed shippingDisplay const cartCountSpan = document.getElementById('cart-count'); const loadingIndicator = document.getElementById('loadingIndicator'); const mainContent = document.getElementById('mainContent'); // Payment method specific elements const paymentMethodRadios = document.querySelectorAll('input[name="paymentMethod"]'); const bankDetailsSection = document.getElementById('bankDetailsSection'); const easyPaisaSection = document.getElementById('easyPaisaSection'); // Mobile menu elements const mobileMenuButton = document.getElementById('mobile-menu-button'); const mobileMenu = document.getElementById('mobile-menu'); const cartCountMobileSpan = document.getElementById('cart-count-mobile'); // --- CART FUNCTIONS --- function getCart() { try { const cart = JSON.parse(localStorage.getItem('cart')) || []; return cart; } catch (e) { console.error("Error parsing cart from localStorage:", e); return []; } } function saveCart(cart) { localStorage.setItem('cart', JSON.stringify(cart)); updateCartDisplay(); } function updateCartDisplay() { const cart = getCart(); const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0); if (cartCountSpan) { cartCountSpan.textContent = totalItems; } if (cartCountMobileSpan) { // Update mobile cart count cartCountMobileSpan.textContent = totalItems; } } function renderCheckoutSummary() { const cart = getCart(); checkoutItemsContainer.innerHTML = ''; let subtotal = 0; if (cart.length === 0) { checkoutItemsContainer.innerHTML = '

Your cart is empty.

'; } else { cart.forEach(item => { const itemTotal = item.price * item.quantity; subtotal += itemTotal; const itemElement = document.createElement('div'); itemElement.className = 'flex justify-between items-center py-2 border-b last:border-b-0'; itemElement.innerHTML = `

${item.name}

Quantity: ${item.quantity}

$${itemTotal.toFixed(2)} `; checkoutItemsContainer.appendChild(itemElement); }); } // Shipping is removed, so total is just subtotal const total = subtotal; subtotalDisplay.textContent = `$${subtotal.toFixed(2)}`; totalAmountDisplay.textContent = `$${total.toFixed(2)}`; } // --- PAYMENT METHOD DISPLAY LOGIC --- function updatePaymentDetailsDisplay() { const selectedMethod = document.querySelector('input[name="paymentMethod"]:checked').value; bankDetailsSection.style.display = 'none'; easyPaisaSection.style.display = 'none'; if (selectedMethod === 'bank') { bankDetailsSection.style.display = 'block'; } else if (selectedMethod === 'easypaisa') { easyPaisaSection.style.display = 'block'; } } // --- FORM SUBMISSION & SUPABASE INSERTION --- checkoutForm.addEventListener('submit', async function (event) { event.preventDefault(); // Prevent default form submission loadingIndicator.classList.remove('hidden'); // Show loading indicator const fullName = document.getElementById('fullName').value; const email = document.getElementById('email').value; const phone = document.getElementById('phone').value; const address = document.getElementById('address').value; const totalAmount = parseFloat(totalAmountDisplay.textContent.replace('$', '')); const cartItems = getCart(); const selectedPaymentMethod = document.querySelector('input[name="paymentMethod"]:checked').value; let transactionId = null; let screenshotFile = null; let screenshotUrl = null; // To store the URL of the uploaded screenshot // Get payment method specific data if (selectedPaymentMethod === 'bank') { transactionId = document.getElementById('bankTransactionId').value; screenshotFile = document.getElementById('bankScreenshot').files[0]; } else if (selectedPaymentMethod === 'easypaisa') { transactionId = document.getElementById('easyPaisaTransactionId').value; screenshotFile = document.getElementById('easyPaisaScreenshot').files[0]; } if (cartItems.length === 0) { alert('Your cart is empty. Please add items before placing an order.'); loadingIndicator.classList.add('hidden'); return; } // Upload screenshot if available if (screenshotFile) { const fileExtension = screenshotFile.name.split('.').pop(); // Ensure unique filename const fileName = `${Date.now()}-${Math.random().toString(36).substring(2, 15)}.${fileExtension}`; const filePath = `transaction-screenshots/${fileName}`; try { // Upload the file to Supabase Storage const { data: uploadData, error: uploadError } = await supabase.storage .from('transaction-screenshots') // Your storage bucket name .upload(filePath, screenshotFile, { cacheControl: '3600', // Cache for 1 hour upsert: false, // Do not overwrite existing files }); if (uploadError) { console.error('Error uploading screenshot:', uploadError); alert('Error uploading screenshot. Please try again.'); loadingIndicator.classList.add('hidden'); return; } // Get the public URL of the uploaded file screenshotUrl = supabase.storage.from('transaction-screenshots').getPublicUrl(filePath).data.publicUrl; console.log('Screenshot uploaded:', screenshotUrl); } catch (uploadException) { console.error('Unexpected error during screenshot upload:', uploadException); alert('An unexpected error occurred during screenshot upload. Please try again.'); loadingIndicator.classList.add('hidden'); return; } } // Prepare product_details as a JSON array of objects const productDetails = cartItems.map(item => ({ id: item.id, name: item.name, quantity: item.quantity, price: item.price })); try { // Insert order into Supabase database const { data, error } = await supabase .from('orders') .insert([ { customer_name: fullName, customer_email: email, customer_phno: phone, customer_address: address, total_amount: totalAmount, payment_method: selectedPaymentMethod, // Save selected payment method transaction_id: transactionId, // Save transaction ID transaction_screenshot_url: screenshotUrl, // Save screenshot URL product_details: productDetails } ]) .select(); // Use .select() to return the inserted data if (error) { console.error('Error saving order to Supabase:', error); alert('There was an error processing your order. Please try again.'); } else { console.log('Order saved successfully:', data); localStorage.removeItem('cart'); // Clear cart after successful order window.location.href = 'thank-you.html?order_success=true'; // Redirect to thank you page } } catch (dbError) { console.error('An unexpected error occurred during database insertion:', dbError); alert('An unexpected error occurred. Please try again.'); } finally { loadingIndicator.classList.add('hidden'); // Hide loading indicator regardless of success/failure } }); // --- EVENT LISTENERS --- paymentMethodRadios.forEach(radio => { radio.addEventListener('change', updatePaymentDetailsDisplay); }); // Mobile menu toggle mobileMenuButton.addEventListener('click', () => { mobileMenu.classList.toggle('hidden'); }); // --- INITIAL SETUP ON PAGE LOAD --- document.addEventListener('DOMContentLoaded', () => { try { renderCheckoutSummary(); updateCartDisplay(); updatePaymentDetailsDisplay(); // Initialize display based on default checked radio } catch (error) { console.error("An error occurred during page setup:", error); } finally { // Hide loading indicator and show main content after a brief delay setTimeout(() => { loadingIndicator.classList.add('hidden'); mainContent.classList.remove('opacity-0'); }, 800); } }); // Smooth scrolling for anchor links (if any are added to checkout.html later) document.querySelectorAll('a[href^="#"]').forEach(anchor => { anchor.addEventListener('click', function (e) { e.preventDefault(); document.querySelector(this.getAttribute('href')).scrollIntoView({ behavior: 'smooth' }); }); });
